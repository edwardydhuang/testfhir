// --- 表單提交 (上傳邏輯) --- document.getElementById('fhirForm').addEventListener('submit', async function(e) { e.preventDefault(); const btn = document.getElementById('submitBtn'); const statusMsg = document.getElementById('statusMsg'); const queryArea = document.getElementById('queryArea'); btn.disabled = true; btn.textContent = '資料上傳中...'; statusMsg.textContent = ''; statusMsg.className = 'status'; queryArea.style.display = 'none'; const age = parseInt(document.getElementById('age').value); const gender = document.getElementById('gender').value; const height = parseFloat(document.getElementById('height').value); const weight = parseFloat(document.getElementById('weight').value); const systolic = parseFloat(document.getElementById('systolic').value); const diastolic = parseFloat(document.getElementById('diastolic').value); const hasHypertension = document.querySelector('input[name="hypertension"]:checked').value === 'yes'; const takingMedication = document.querySelector('input[name="medication"]:checked').value === 'yes'; const patientUUID = "urn:uuid:" + generateUUID(); const currentYear = new Date().getFullYear(); const birthYear = currentYear - age; const currentTime = new Date().toISOString(); // 寫入自訂 Tag 到 Meta 中 const commonMeta = { meta: { tag: [SYSTEM_TAG] } }; const entries = []; // 1. Patient entries.push({ fullUrl: patientUUID, // Patient 本身的 UUID request: { method: "POST", url: "Patient" }, resource: { resourceType: "Patient", ...commonMeta, gender: gender, birthDate: `${birthYear}-01-01` } }); const baseObservation = { resourceType: "Observation", ...commonMeta, status: "final", subject: { reference: patientUUID }, effectiveDateTime: currentTime }; // 2. Height (注意：加上了 fullUrl) entries.push({ fullUrl: "urn:uuid:" + generateUUID(), request: { method: "POST", url: "Observation" }, resource: { ...baseObservation, category: [{ coding: [{ system: "http://terminology.hl7.org/CodeSystem/observation-category", code: "vital-signs" }] }], code: { coding: [{ system: "http://loinc.org", code: "8302-2", display: "Body height" }] }, valueQuantity: { value: height, unit: "cm", system: "http://unitsofmeasure.org", code: "cm" } } }); // 3. Weight (注意：加上了 fullUrl) entries.push({ fullUrl: "urn:uuid:" + generateUUID(), request: { method: "POST", url: "Observation" }, resource: { ...baseObservation, category: [{ coding: [{ system: "http://terminology.hl7.org/CodeSystem/observation-category", code: "vital-signs" }] }], code: { coding: [{ system: "http://loinc.org", code: "29463-7", display: "Body weight" }] }, valueQuantity: { value: weight, unit: "kg", system: "http://unitsofmeasure.org", code: "kg" } } }); // 4. Blood Pressure (注意：加上了 fullUrl) entries.push({ fullUrl: "urn:uuid:" + generateUUID(), request: { method: "POST", url: "Observation" }, resource: { ...baseObservation, category: [{ coding: [{ system: "http://terminology.hl7.org/CodeSystem/observation-category", code: "vital-signs" }] }], code: { coding: [{ system: "http://loinc.org", code: "85354-9", display: "Blood pressure panel with all children optional" }] }, component: [ { code: { coding: [{ system: "http://loinc.org", code: "8480-6", display: "Systolic blood pressure" }] }, valueQuantity: { value: systolic, unit: "mmHg", system: "http://unitsofmeasure.org", code: "mm[Hg]" } }, { code: { coding: [{ system: "http://loinc.org", code: "8462-4", display: "Diastolic blood pressure" }] }, valueQuantity: { value: diastolic, unit: "mmHg", system: "http://unitsofmeasure.org", code: "mm[Hg]" } } ] } }); // 5. Condition if (hasHypertension) { entries.push({ fullUrl: "urn:uuid:" + generateUUID(), request: { method: "POST", url: "Condition" }, resource: { resourceType: "Condition", ...commonMeta, clinicalStatus: { coding: [{ system: "http://terminology.hl7.org/CodeSystem/condition-clinical", code: "active" }] }, code: { coding: [{ system: "http://snomed.info/sct", code: "38341003", display: "Hypertension" }] }, subject: { reference: patientUUID } } }); } // 6. MedicationStatement if (takingMedication) { entries.push({ fullUrl: "urn:uuid:" + generateUUID(), request: { method: "POST", url: "MedicationStatement" }, resource: { resourceType: "MedicationStatement", ...commonMeta, status: "active", medicationCodeableConcept: { text: "降血壓藥物 (Antihypertensive medication)" }, subject: { reference: patientUUID } } }); } const fhirBundle = { resourceType: "Bundle", type: "transaction", entry: entries }; try { const response = await fetch(FHIR_BASE_URL, { method: 'POST', headers: { 'Content-Type': 'application/fhir+json' }, body: JSON.stringify(fhirBundle) }); if (response.ok) { const result = await response.json(); statusMsg.textContent = '資料寫入完成！'; statusMsg.className = 'status success'; const patientLocation = result.entry[0].response.location; const patientId = patientLocation.split('/')[1]; await fetchAndDisplayVitals(patientId); } else { // --- 增強型錯誤捕捉：解析 FHIR 的 OperationOutcome --- const errorResult = await response.json(); console.error("伺服器錯誤回覆:", errorResult); let errorText = `未知伺服器錯誤`; if (errorResult.resourceType === 'OperationOutcome' && errorResult.issue) { // 抓取伺服器回傳的具體診斷訊息 (Diagnostics) errorText = errorResult.issue.map(i => i.diagnostics || i.code).join(' ; '); } statusMsg.textContent = `上傳失敗 (${response.status})：${errorText}`; statusMsg.className = 'status error'; } } catch (error) { statusMsg.textContent = `網路或系統發生錯誤：${error.message}`; statusMsg.className = 'status error'; } finally { btn.disabled = false; btn.textContent = '打包、上傳並查詢本筆資料'; } });
